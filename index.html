# CareerHub / Job Portal - Production Django Architecture
# Live URL: https://careerhub-yash.onrender.com

from django.db import models
from django.shortcuts import render

class JobOpening(models.Model):
    title = models.CharField(max_length=200)
    company_name = models.CharField(max_length=150)
    location = models.CharField(max_length=100)
    experience_level = models.CharField(max_length=50)
    is_active = models.BooleanField(default=True)

def portal_feed_view(request):
    openings = JobOpening.objects.filter(is_active=True)
    return render(request, "jobs/feed.html", {"jobs": openings})
